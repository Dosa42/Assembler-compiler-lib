cmake_minimum_required(VERSION 3.22)
project(brokkr_engine VERSION 2.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_WITH_INSTALL_RPATH ON)
option(BROKKR_STATIC "Build a static companion and C API archive" OFF)
set(UPSTREAM ${CMAKE_CURRENT_SOURCE_DIR}/upstream/original)
find_path(LIBUSB_INCLUDE_DIR libusb.h PATH_SUFFIXES libusb-1.0 REQUIRED)
find_library(LIBUSB_LIBRARY usb-1.0 REQUIRED)
set(CORE_SOURCES
  src/core/thread_pool.cpp
  src/io/random_access.cpp src/io/sequential_reader.cpp src/io/tar.cpp src/io/source.cpp src/io/lz4_frame.cpp
  src/third_party/md5/md5.c src/third_party/xxhash/xxhash.c src/third_party/xxhash/xxhash_dispatch.c src/third_party/lz4/lz4.c
  src/protocol/odin/odin_cmd.cpp src/protocol/odin/pit.cpp src/protocol/odin/flash.cpp
  src/protocol/odin/group_flasher.cpp src/protocol/odin/pit_transfer.cpp
  src/app/md5_xxh3_cache.cpp src/app/md5_verify.cpp
  src/platform/android/app_dirs.cpp src/platform/android/libusb_transport.cpp)
list(TRANSFORM CORE_SOURCES PREPEND "${UPSTREAM}/")
if(BROKKR_STATIC)
  add_library(brokkr-engine-api STATIC src/engine.cpp usb/brokkr_usb_client.c ${CORE_SOURCES})
else()
  add_library(brokkr-engine-api SHARED src/engine.cpp usb/brokkr_usb_client.c ${CORE_SOURCES})
endif()
target_include_directories(brokkr-engine-api PRIVATE ${UPSTREAM}/src ${UPSTREAM}/src-jni/shim ${LIBUSB_INCLUDE_DIR} usb PUBLIC include)
target_compile_definitions(brokkr-engine-api PRIVATE BROKKR_PLATFORM_ANDROID _FILE_OFFSET_BITS=64)
target_compile_options(brokkr-engine-api PRIVATE -O2 -ffunction-sections -fdata-sections)
target_link_options(brokkr-engine-api PRIVATE -Wl,--gc-sections -Wl,--no-undefined)
find_package(Threads REQUIRED)
if(BROKKR_STATIC)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(USB_STATIC REQUIRED libusb-1.0)
  target_link_libraries(brokkr-engine-api PRIVATE ${USB_STATIC_STATIC_LIBRARIES} Threads::Threads)
  target_link_directories(brokkr-engine-api PRIVATE ${USB_STATIC_STATIC_LIBRARY_DIRS})
else()
  target_link_libraries(brokkr-engine-api PRIVATE ${LIBUSB_LIBRARY} Threads::Threads)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Android")
  target_link_libraries(brokkr-engine-api PRIVATE log)
endif()
set_target_properties(brokkr-engine-api PROPERTIES OUTPUT_NAME brokkr-engine INSTALL_RPATH "$ORIGIN")
if(NOT BROKKR_STATIC)
  set_target_properties(brokkr-engine-api PROPERTIES SOVERSION 1)
endif()
add_executable(brokkr-engine src/main.cpp)
target_link_libraries(brokkr-engine PRIVATE brokkr-engine-api)
if(BROKKR_STATIC)
  target_link_options(brokkr-engine PRIVATE -static)
endif()
set_target_properties(brokkr-engine PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS brokkr-engine brokkr-engine-api RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(FILES include/brokkr_engine.h DESTINATION include)
